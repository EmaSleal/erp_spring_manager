```java
package com.erp.whatsorders.model;

import jakarta.persistence.*;
import jakarta.validation.constraints.*;
import lombok.Data;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;

/**
 * Representa una nómina (planilla) de la empresa.
 * Puede ser quincenal o mensual.
 */
@Entity
@Table(name = "nominas")
@Data
public class Nomina {
    
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    
    /**
     * Período de la nómina (ej: 2026-01, 2026-01-Q1)
     */
    @Column(nullable = false, unique = true, length = 20)
    @NotBlank(message = "Período es obligatorio")
    private String periodo;
    
    /**
     * Tipo de nómina
     */
    @Enumerated(EnumType.STRING)
    @Column(nullable = false, length = 20)
    private TipoNomina tipo;
    
    /**
     * Fecha desde
     */
    @Column(nullable = false)
    @NotNull(message = "Fecha desde es obligatoria")
    private LocalDate fechaDesde;
    
    /**
     * Fecha hasta
     */
    @Column(nullable = false)
    @NotNull(message = "Fecha hasta es obligatoria")
    private LocalDate fechaHasta;
    
    /**
     * Fecha de pago
     */
    @Column(nullable = false)
    @NotNull(message = "Fecha de pago es obligatoria")
    private LocalDate fechaPago;
    
    /**
     * Estado de la nómina
     */
    @Enumerated(EnumType.STRING)
    @Column(nullable = false, length = 20)
    private EstadoNomina estado = EstadoNomina.BORRADOR;
    
    /**
     * Total de salarios brutos
     */
    @Column(precision = 12, scale = 2)
    private BigDecimal totalSalariosBrutos = BigDecimal.ZERO;
    
    /**
     * Total de deducciones CCSS obrero
     */
    @Column(precision = 12, scale = 2)
    private BigDecimal totalDeduccionesCCSSObrero = BigDecimal.ZERO;
    
    /**
     * Total de deducciones INS
     */
    @Column(precision = 12, scale = 2)
    private BigDecimal totalDeduccionesINS = BigDecimal.ZERO;
    
    /**
     * Total de impuesto sobre la renta
     */
    @Column(precision = 12, scale = 2)
    private BigDecimal totalImpuestoRenta = BigDecimal.ZERO;
    
    /**
     * Total de otras deducciones (préstamos, embargos)
     */
    @Column(precision = 12, scale = 2)
    private BigDecimal totalOtrasDeducciones = BigDecimal.ZERO;
    
    /**
     * Total de salarios netos
     */
    @Column(precision = 12, scale = 2)
    private BigDecimal totalSalariosNetos = BigDecimal.ZERO;
    
    /**
     * Total de cargas patronales CCSS
     */
    @Column(precision = 12, scale = 2)
    private BigDecimal totalCargasPatronales = BigDecimal.ZERO;
    
    /**
     * Total de aguinaldo acumulado
     */
    @Column(precision = 12, scale = 2)
    private BigDecimal totalAguinaldoAcumulado = BigDecimal.ZERO;
    
    /**
     * Total de cesantía acumulada
     */
    @Column(precision = 12, scale = 2)
    private BigDecimal totalCesantiaAcumulada = BigDecimal.ZERO;
    
    /**
     * Total general a pagar
     */
    @Column(precision = 12, scale = 2)
    private BigDecimal totalGeneral = BigDecimal.ZERO;
    
    /**
     * Observaciones
     */
    @Column(columnDefinition = "TEXT")
    private String observaciones;
    
    /**
     * Detalles de la nómina (un detalle por empleado)
     */
    @OneToMany(mappedBy = "nomina", cascade = CascadeType.ALL, orphanRemoval = true)
    private List<DetalleNomina> detalles = new ArrayList<>();
    
    /**
     * Usuario que procesó la nómina
     */
    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "procesado_por_id")
    private Usuario procesadoPor;
    
    /**
     * Fecha de procesamiento
     */
    @Column
    private LocalDateTime fechaProcesamiento;
    
    @Column(nullable = false, updatable = false)
    private LocalDateTime createdAt;
    
    @Column(nullable = false)
    private LocalDateTime updatedAt;
    
    @PrePersist
    protected void onCreate() {
        createdAt = LocalDateTime.now();
        updatedAt = LocalDateTime.now();
    }
    
    @PreUpdate
    protected void onUpdate() {
        updatedAt = LocalDateTime.now();
    }
    
    /**
     * Calcula totales de la nómina.
     */
    public void calcularTotales() {
        totalSalariosBrutos = detalles.stream()
            .map(DetalleNomina::getSalarioBruto)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        
        totalDeduccionesCCSSObrero = detalles.stream()
            .map(DetalleNomina::getDeduccionCCSSObrero)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        
        totalDeduccionesINS = detalles.stream()
            .map(DetalleNomina::getDeduccionINS)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        
        totalImpuestoRenta = detalles.stream()
            .map(DetalleNomina::getImpuestoRenta)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        
        totalOtrasDeducciones = detalles.stream()
            .map(DetalleNomina::getOtrasDeducciones)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        
        totalSalariosNetos = detalles.stream()
            .map(DetalleNomina::getSalarioNeto)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        
        totalCargasPatronales = detalles.stream()
            .map(DetalleNomina::getCargaPatronalCCSS)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        
        totalAguinaldoAcumulado = detalles.stream()
            .map(DetalleNomina::getAguinaldoAcumulado)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        
        totalCesantiaAcumulada = detalles.stream()
            .map(DetalleNomina::getCesantiaAcumulada)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        
        totalGeneral = totalSalariosNetos;
    }
}

enum TipoNomina {
    MENSUAL,
    QUINCENAL,
    SEMANAL,
    ESPECIAL
}

enum EstadoNomina {
    BORRADOR,      // En construcción
    CALCULADA,     // Cálculos realizados
    APROBADA,      // Aprobada para pago
    PAGADA,        // Ya se pagó
    CONTABILIZADA, // Registrada en contabilidad
    ANULADA        // Anulada
}
```

- [ ] **1.1.6** Crear servicio de cálculos de nómina

```java
package com.erp.whatsorders.service;

import com.erp.whatsorders.model.*;
import com.erp.whatsorders.util.NominaConstantes;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import java.math.BigDecimal;
import java.math.RoundingMode;

/**
 * Servicio para cálculos de nómina según normativa de Costa Rica.
 */
@Service
@Slf4j
public class CalculosNominaService {
    
    /**
     * Calcula el salario bruto total de un empleado.
     */
    public BigDecimal calcularSalarioBruto(DetalleNomina detalle) {
        BigDecimal salarioBase = detalle.getSalarioBase();
        BigDecimal totalHoras = BigDecimal.ZERO;
        
        // Calcular valor hora
        BigDecimal valorHora = salarioBase.divide(
            NominaConstantes.HORAS_MENSUALES_ORDINARIAS, 
            2, 
            RoundingMode.HALF_UP
        );
        
        // Horas ordinarias
        if (detalle.getHorasOrdinarias() != null) {
            totalHoras = totalHoras.add(
                detalle.getHorasOrdinarias().multiply(valorHora)
            );
        }
        
        // Horas extra 50%
        if (detalle.getHorasExtra50() != null) {
            totalHoras = totalHoras.add(
                detalle.getHorasExtra50()
                    .multiply(valorHora)
                    .multiply(NominaConstantes.RECARGO_HORAS_EXTRA_50)
            );
        }
        
        // Horas extra doble
        if (detalle.getHorasExtraDoble() != null) {
            totalHoras = totalHoras.add(
                detalle.getHorasExtraDoble()
                    .multiply(valorHora)
                    .multiply(NominaConstantes.RECARGO_HORAS_EXTRA_DOBLE)
            );
        }
        
        // Horas nocturnas
        if (detalle.getHorasNocturnas() != null) {
            totalHoras = totalHoras.add(
                detalle.getHorasNocturnas()
                    .multiply(valorHora)
                    .multiply(NominaConstantes.RECARGO_HORAS_NOCTURNAS)
            );
        }
        
        // Sumar bonificaciones, comisiones, etc.
        BigDecimal bruto = salarioBase.add(totalHoras);
        bruto = bruto.add(detalle.getBonificaciones() != null ? detalle.getBonificaciones() : BigDecimal.ZERO);
        bruto = bruto.add(detalle.getComisiones() != null ? detalle.getComisiones() : BigDecimal.ZERO);
        bruto = bruto.add(detalle.getOtrosIngresos() != null ? detalle.getOtrosIngresos() : BigDecimal.ZERO);
        
        return bruto.setScale(2, RoundingMode.HALF_UP);
    }
    
    /**
     * Calcula deducción CCSS obrero (9.34%).
     */
    public BigDecimal calcularDeduccionCCSSObrero(BigDecimal salarioBruto) {
        return salarioBruto
            .multiply(NominaConstantes.CCSS_OBRERO_PORCENTAJE)
            .divide(new BigDecimal("100"), 2, RoundingMode.HALF_UP);
    }
    
    /**
     * Calcula deducción INS (~1%).
     */
    public BigDecimal calcularDeduccionINS(BigDecimal salarioBruto) {
        return salarioBruto
            .multiply(NominaConstantes.INS_RIESGOS_PORCENTAJE)
            .divide(new BigDecimal("100"), 2, RoundingMode.HALF_UP);
    }
    
    /**
     * Calcula impuesto sobre la renta (progresivo).
     */
    public BigDecimal calcularImpuestoRenta(BigDecimal salarioBruto) {
        // Salario bruto menos deducciones no gravables (CCSS + INS)
        BigDecimal deduccionCCSS = calcularDeduccionCCSSObrero(salarioBruto);
        BigDecimal deduccionINS = calcularDeduccionINS(salarioBruto);
        BigDecimal salarioGravable = salarioBruto.subtract(deduccionCCSS).subtract(deduccionINS);
        
        BigDecimal impuesto = BigDecimal.ZERO;
        
        // Tramo 1: Exento hasta ₡941,000
        if (salarioGravable.compareTo(NominaConstantes.ImpuestoRenta.TRAMO_1_HASTA) <= 0) {
            return BigDecimal.ZERO;
        }
        
        // Tramo 2: 10% de 941,001 a 1,381,000
        if (salarioGravable.compareTo(NominaConstantes.ImpuestoRenta.TRAMO_2_HASTA) <= 0) {
            BigDecimal excedente = salarioGravable.subtract(NominaConstantes.ImpuestoRenta.TRAMO_1_HASTA);
            impuesto = excedente
                .multiply(NominaConstantes.ImpuestoRenta.TRAMO_2_PORCENTAJE)
                .divide(new BigDecimal("100"), 2, RoundingMode.HALF_UP);
            return impuesto;
        }
        
        // Tramo 3: 15% de 1,381,001 a 2,423,000
        if (salarioGravable.compareTo(NominaConstantes.ImpuestoRenta.TRAMO_3_HASTA) <= 0) {
            // Impuesto del tramo 2 completo
            BigDecimal tramo2 = NominaConstantes.ImpuestoRenta.TRAMO_2_HASTA
                .subtract(NominaConstantes.ImpuestoRenta.TRAMO_1_HASTA)
                .multiply(NominaConstantes.ImpuestoRenta.TRAMO_2_PORCENTAJE)
                .divide(new BigDecimal("100"), 2, RoundingMode.HALF_UP);
            
            // Impuesto del excedente en tramo 3
            BigDecimal excedente = salarioGravable.subtract(NominaConstantes.ImpuestoRenta.TRAMO_2_HASTA);
            BigDecimal tramo3 = excedente
                .multiply(NominaConstantes.ImpuestoRenta.TRAMO_3_PORCENTAJE)
                .divide(new BigDecimal("100"), 2, RoundingMode.HALF_UP);
            
            return tramo2.add(tramo3);
        }
        
        // Tramo 4: 20% de 2,423,001 a 4,846,000
        if (salarioGravable.compareTo(NominaConstantes.ImpuestoRenta.TRAMO_4_HASTA) <= 0) {
            // Sumar impuestos de tramos anteriores
            BigDecimal tramosAnteriores = calcularImpuestoTramosAnteriores(
                NominaConstantes.ImpuestoRenta.TRAMO_3_HASTA
            );
            
            BigDecimal excedente = salarioGravable.subtract(NominaConstantes.ImpuestoRenta.TRAMO_3_HASTA);
            BigDecimal tramo4 = excedente
                .multiply(NominaConstantes.ImpuestoRenta.TRAMO_4_PORCENTAJE)

            return tramosAnteriores.add(tramo4);

        // Tramo 5: 25% de 4,846,001 en adelante

        BigDecimal excedente = salarioGravable.subtract(NominaConstantes.ImpuestoRenta.TRAMO_4_HASTA);
        BigDecimal tramo5 = excedente
            .multiply(NominaConstantes.ImpuestoRenta.TRAMO_5_PORCENTAJE)
            .divide(new BigDecimal("100"), 2, RoundingMode.HALF_UP);

    /**
     * Calcula todas las deducciones y el salario neto.
     */
    public void calcularDetalleCompleto(DetalleNomina detalle) {
        // 1. Calcular salario bruto
        BigDecimal salarioBruto = calcularSalarioBruto(detalle);
        detalle.setSalarioBruto(salarioBruto);
        
        // 2. Calcular deducciones
        BigDecimal ccssObrero = calcularDeduccionCCSSObrero(salarioBruto);
        detalle.setDeduccionCCSSObrero(ccssObrero);
        
        BigDecimal ins = calcularDeduccionINS(salarioBruto);
        detalle.setDeduccionINS(ins);
        
        BigDecimal renta = calcularImpuestoRenta(salarioBruto);
        detalle.setImpuestoRenta(renta);
        
        // Total deducciones
        BigDecimal totalDeducciones = ccssObrero
            .add(ins)
            .add(renta)
            .add(detalle.getPrestamos() != null ? detalle.getPrestamos() : BigDecimal.ZERO)
            .add(detalle.getEmbargos() != null ? detalle.getEmbargos() : BigDecimal.ZERO)
            .add(detalle.getOtrasDeducciones() != null ? detalle.getOtrasDeducciones() : BigDecimal.ZERO);
        detalle.setTotalDeducciones(totalDeducciones);
        
        // 3. Calcular salario neto
        BigDecimal salarioNeto = salarioBruto.subtract(totalDeducciones);
        detalle.setSalarioNeto(salarioNeto);
        
        // 4. Calcular cargas patronales
        BigDecimal cargaPatronalCCSS = calcularCargaPatronalCCSS(salarioBruto);
        detalle.setCargaPatronalCCSS(cargaPatronalCCSS);
        
        BigDecimal cargaPatronalINS = calcularDeduccionINS(salarioBruto); // Mismo cálculo
        detalle.setCargaPatronalINS(cargaPatronalINS);
        
        // 5. Calcular prestaciones
        BigDecimal aguinaldo = calcularAguinaldo(salarioBruto);
        detalle.setAguinaldoAcumulado(aguinaldo);
        
        BigDecimal cesantia = calcularCesantia(salarioBruto);
        detalle.setCesantiaAcumulada(cesantia);
        
        log.debug("Detalle calculado - Empleado: {}, Bruto: {}, Neto: {}",
            detalle.getEmpleado().getNombreCompleto(),
            salarioBruto,
            salarioNeto
        );
    }
```

```java
@Service
@RequiredArgsConstructor
@Slf4j
public class NominaService {
    
    private final NominaRepository nominaRepository;
    private final DetalleNominaRepository detalleNominaRepository;
    private final EmpleadoRepository empleadoRepository;
    private final CalculosNominaService calculosNominaService;
    
    /**
     * Crea una nueva nómina para el período especificado.
     */
    @Transactional
    public NominaDTO crear(String periodo, TipoNomina tipo, LocalDate fechaDesde, 
                           LocalDate fechaHasta, LocalDate fechaPago) {
        // Validar que no exista nómina para ese período
        if (nominaRepository.findByPeriodo(periodo).isPresent()) {
            throw new BusinessException("Ya existe una nómina para el período: " + periodo);
        }
        
        Nomina nomina = new Nomina();
        nomina.setPeriodo(periodo);
        nomina.setTipo(tipo);
        nomina.setFechaDesde(fechaDesde);
        nomina.setFechaHasta(fechaHasta);
        nomina.setFechaPago(fechaPago);
        nomina.setEstado(EstadoNomina.BORRADOR);
        
        // Crear detalle para cada empleado activo
        List<Empleado> empleadosActivos = empleadoRepository.findByActivoTrue();
        
        for (Empleado empleado : empleadosActivos) {
            DetalleNomina detalle = new DetalleNomina();
            detalle.setNomina(nomina);
            detalle.setEmpleado(empleado);
            detalle.setSalarioBase(empleado.getSalarioBruto());
            
            nomina.getDetalles().add(detalle);
        }
        
        nomina = nominaRepository.save(nomina);
        
        log.info("Nómina creada: {} con {} empleados", periodo, empleadosActivos.size());
        
        return toDTO(nomina);
    }
    
    /**
     * Calcula la nómina completa.
     */
    @Transactional
    public void calcular(Long nominaId) {
        Nomina nomina = nominaRepository.findById(nominaId)
            .orElseThrow(() -> new BusinessException("Nómina no encontrada"));
        
        if (nomina.getEstado() != EstadoNomina.BORRADOR) {
            throw new BusinessException("Solo se pueden calcular nóminas en borrador");
        }
        
        // Calcular cada detalle
        for (DetalleNomina detalle : nomina.getDetalles()) {
            calculosNominaService.calcularDetalleCompleto(detalle);
        }
        
        // Calcular totales de la nómina
        nomina.calcularTotales();
        nomina.setEstado(EstadoNomina.CALCULADA);
        
        nominaRepository.save(nomina);
        
        log.info("Nómina calculada: {} - Total a pagar: ₡{}", 
            nomina.getPeriodo(), nomina.getTotalSalariosNetos());
    }
    
    /**
     * Aprueba la nómina para pago.
     */
    @Transactional
    public void aprobar(Long nominaId, Usuario usuario) {
        Nomina nomina = nominaRepository.findById(nominaId)
            .orElseThrow(() -> new BusinessException("Nómina no encontrada"));
        
        if (nomina.getEstado() != EstadoNomina.CALCULADA) {
            throw new BusinessException("Solo se pueden aprobar nóminas calculadas");
        }
        
        nomina.setEstado(EstadoNomina.APROBADA);
        nomina.setProcesadoPor(usuario);
        nomina.setFechaProcesamiento(LocalDateTime.now());
        
        nominaRepository.save(nomina);
        
        log.info("Nómina aprobada: {} por {}", nomina.getPeriodo(), usuario.getEmail());
    }
    
    private NominaDTO toDTO(Nomina nomina) {
        NominaDTO dto = new NominaDTO();
        dto.setId(nomina.getId());
        dto.setPeriodo(nomina.getPeriodo());
        dto.setTipo(nomina.getTipo());
        dto.setFechaDesde(nomina.getFechaDesde());
        dto.setFechaHasta(nomina.getFechaHasta());
        dto.setFechaPago(nomina.getFechaPago());
        dto.setEstado(nomina.getEstado());
        dto.setTotalSalariosBrutos(nomina.getTotalSalariosBrutos());
        dto.setTotalSalariosNetos(nomina.getTotalSalariosNetos());
        dto.setCantidadEmpleados(nomina.getDetalles().size());
        return dto;
    }
```
